    // 画面遷移用関数
    function showPage(pageId) {
      const pages = ['top', 'schedule', 'log', 'vital', 'race', 'ct', 'bank'];
      pages.forEach(p => {
        document.getElementById(`page-${p}`).style.display = (p === pageId) ? 'block' : 'none';
      });
      window.scrollTo(0, 0); // 画面上部へスクロール
    }

    // データ保存時の共通アラート＆トップ戻り
    function saveData(name) {
      alert(`${name} データを保存しました！トップページへ戻ります。`);
      showPage('top');
    }

    // 練習日誌用 RPEテキスト更新
    function updateRpeText(val) {
      const labels = { 1:"1 / 10 (非常に楽)", 5:"5 / 10 (普通)", 7:"7 / 10 (かなりきつい)", 10:"10 / 10 (オールアウト)" };
      document.getElementById('rpe-text').innerText = labels[val] || `${val} / 10`;
    }

    function addRep(btn) {
      const repList = btn.previousElementSibling;
      const count = repList.querySelectorAll('.row').length + 1;
      const div = document.createElement('div');
      div.className = 'row';
      div.innerHTML = `<div class="rep-label">${count}本目</div><input type="number" step="0.01" placeholder="タイム (秒) または 記録 (m)" style="margin:0;">`;
      repList.appendChild(div);
    }

    function addMenuBlock() {
      const container = document.getElementById('menu-container');
      const div = document.createElement('div');
      div.className = 'menu-block';
      div.style = "background:#252525; padding:12px; border-radius:8px; margin-bottom:12px; border-left:4px solid #00ff88;";
      div.innerHTML = `<div style="font-size:12px; color:#aaa; font-weight:bold; margin-bottom:6px;">追加メニュー</div><select><option>100m</option><option>走幅跳 (全助走)</option></select><div class="rep-list"><div class="row"><div class="rep-label">1本目</div><input type="number" step="0.01" placeholder="タイム / 記録" style="margin:0;"></div></div><button class="btn-add" onclick="addRep(this)" style="margin-top:8px;">＋ 本数を追加</button>`;
      container.appendChild(div);
    }

    // 部位タグ切替
    function toggleTag(el) { el.classList.toggle('selected'); }

    // 跳躍クイック入力
    function setQuickVal(btn, val) { btn.parentElement.querySelector('.jump-val').value = val; }

    function addFinalJumps() {
      const container = document.getElementById('jump-rows');
      for (let i = 4; i <= 6; i++) {
        const div = document.createElement('div');
        div.className = 'row';
        div.innerHTML = `<div class="rep-label">${i}本目</div><input type="text" placeholder="記録" class="jump-val" style="margin:0;"><button class="tag-btn" onclick="setQuickVal(this, 'F')">F</button><button class="tag-btn" onclick="setQuickVal(this, '-')">-</button><input type="number" step="0.1" placeholder="風" style="margin:0;">`;
        container.appendChild(div);
      }
      document.getElementById('btn-add-final').style.display = 'none';
    }

    // 試合結果 種別切替
    function switchRaceFormType(type) {
      document.getElementById('form-horizontal-jump').style.display = (type === 'horizontal-jump') ? 'block' : 'none';
      document.getElementById('form-high-jump').style.display = (type === 'high-jump') ? 'block' : 'none';
      document.getElementById('form-sprint').style.display = (type === 'sprint') ? 'block' : 'none';
    }

      document.getElementById('detail-mode').style.display = (mode === 'detail') ? 'block' : 'none';
    }
